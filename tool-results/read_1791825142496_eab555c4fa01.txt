   812→export function translate(locale: Locale, key: string, vars?: Record<string, string | number>): string {
   813→  const dict = dictionaries[locale] || dictionaries.en
   814→  let s = dict[key] ?? dictionaries.en[key] ?? key
   815→  if (vars) {
   816→    for (const [k, v] of Object.entries(vars)) {
   817→      s = s.replace(new RegExp(`\\{${k}\\}`, 'g'), String(v))
   818→    }
   819→  }
   820→  return s
   821→}
