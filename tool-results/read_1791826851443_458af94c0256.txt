   174→function CountTooltip({
   175→  active,
   176→  payload,
   177→  label,
   178→  countLabel,
   179→  num,
   180→}: {
   181→  active?: boolean
   182→  payload?: { value?: number; color?: string }[]
   183→  label?: string | number
   184→  countLabel: string
   185→  num: (v: number) => string
   186→}) {
   187→  if (!active || !payload || payload.length === 0) return null
   188→  return (
   189→    <div className="rounded-md border bg-popover p-3 text-xs shadow-md">
   190→      {label !== undefined && label !== '' && (
   191→        <p className="mb-1 flex items-center gap-1.5 font-medium text-foreground">
   192→          <span
   193→            className="inline-block h-2.5 w-2.5 rounded-full"
   194→            style={{ backgroundColor: payload[0]?.color || DEFAULT_COLOR }}
   195→          />
   196→          {label}
   197→        </p>
   198→      )}
   199→      <p className="tabular-nums text-muted-foreground">
   200→        {countLabel}:{' '}
   201→        <span className="font-medium text-foreground">
   202→          {num(Number(payload[0]?.value ?? 0))}
   203→        </span>
   204→      </p>
   205→    </div>
   206→  )
   207→}
