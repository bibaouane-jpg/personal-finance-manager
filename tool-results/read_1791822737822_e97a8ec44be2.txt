   148→function clampPage(p: number, totalPages: number): number {
   149→  if (p < 1) return 1
   150→  if (p > totalPages) return totalPages
   151→  return p
   152→}
   153→
   154→function buildPageList(current: number, totalPages: number): (number | 'ellipsis')[] {
   155→  if (totalPages <= 7) {
   156→    return Array.from({ length: totalPages }, (_, i) => i + 1)
   157→  }
   158→  const pages: (number | 'ellipsis')[] = [1]
   159→  const left = Math.max(2, current - 1)
   160→  const right = Math.min(totalPages - 1, current + 1)
   161→  if (left > 2) pages.push('ellipsis')
   162→  for (let i = left; i <= right; i++) pages.push(i)
   163→  if (right < totalPages - 1) pages.push('ellipsis')
   164→  pages.push(totalPages)
   165→  return pages
   166→}
