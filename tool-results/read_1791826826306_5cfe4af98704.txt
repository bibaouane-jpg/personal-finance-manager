   309→   309→function formatMonthLabel(month: string, locale: 'en' | 'fr'): string {
   310→   310→  // month comes as "YYYY-MM"
   311→   311→  const d = new Date(`${month}-01T00:00:00Z`)
   312→   312→  if (isNaN(d.getTime())) return month
   313→   313→  return new Intl.DateTimeFormat(locale === 'fr' ? 'fr-FR' : 'en-US', {
   314→   314→    month: 'short',
   315→   315→    year: '2-digit',
   316→   316→    timeZone: 'UTC',
   317→   317→  }).format(d)
   318→   318→}
   319→   319→
   320→   320→function formatTrendDate(iso: string, locale: 'en' | 'fr'): string {
   321→   321→  const d = new Date(iso)
   322→   322→  if (isNaN(d.getTime())) return iso
   323→   323→  return new Intl.DateTimeFormat(locale === 'fr' ? 'fr-FR' : 'en-US', {
   324→   324→    month: 'short',
   325→   325→    day: 'numeric',
   326→   326→  }).format(d)
   327→   327→}
   328→   328→
   329→   329→function formatYearLabel(year: number, locale: 'en' | 'fr'): string {
   330→   330→  return new Intl.DateTimeFormat(locale === 'fr' ? 'fr-FR' : 'en-US', {
   331→   331→    year: 'numeric',
   332→   332→  }).format(new Date(Date.UTC(year, 0, 1)))
   333→   333→}

   335→   335→function asNumber(v: string | number | null | undefined): number {
   336→   336→  if (v === null || v === undefined) return 0
   337→   337→  const n = typeof v === 'number' ? v : Number(v)
   338→   338→  return Number.isFinite(n) ? n : 0
   339→   339→}
