   133→const CREDIT_STATUSES: CreditStatus[] = ['active', 'completed', 'on_hold']
   134→
   135→function todayLocal(): string {
   136→  return toLocalDateString(new Date())
   137→}
   138→
   139→function startOfToday(): Date {
   140→  const d = new Date()
   141→  d.setHours(0, 0, 0, 0)
   142→  return d
   143→}
   144→
   145→/** Map a repayment progress percent to a credit-card color palette.
   146→ *  <50% red, 50-<70% orange, 70-<90% dark green, 90-100% green. */
   147→function creditCardColor(progressPct: number): {
   148→  stripe: string
   149→  text: string
   150→  tint: string
   151→} {
   152→  if (progressPct < 50) {
   153→    return { stripe: '#ef4444', text: '#b91c1c', tint: 'rgba(239,68,68,0.08)' }
   154→  }
   155→  if (progressPct < 70) {
   156→    return { stripe: '#f97316', text: '#c2410c', tint: 'rgba(249,115,22,0.08)' }
   157→  }
   158→  if (progressPct < 90) {
   159→    return { stripe: '#15803d', text: '#15803d', tint: 'rgba(21,128,61,0.10)' }
   160→  }
   161→  return { stripe: '#22c55e', text: '#15803d', tint: 'rgba(34,197,94,0.10)' }
   162→}
   163→
   164→/** Classify an installment segment color for the schedule grid. */
   165→function installmentSegment(inst: CreditInstallment): {
   166→  bg: string
   167→  border: string
   168→  text: string
   169→  kind: 'paid' | 'overdue' | 'pending'
   170→} {
   171→  if (inst.status === 'paid') {
   172→    return {
   173→      bg: 'rgba(34,197,94,0.15)',
   174→      border: 'rgba(34,197,94,0.55)',
   175→      text: '#15803d',
   176→      kind: 'paid',
   177→    }
   178→  }
   179→  const due = new Date(inst.dueDate)
   180→  if (!isNaN(due.getTime()) && due < startOfToday()) {
   181→    return {
   182→      bg: 'rgba(239,68,68,0.15)',
   183→      border: 'rgba(239,68,68,0.55)',
   184→      text: '#b91c1c',
   185→      kind: 'overdue',
   186→    }
   187→  }
   188→  return {
   189→    bg: 'rgba(245,158,11,0.15)',
   190→    border: 'rgba(245,158,11,0.45)',
   191→    text: '#a16207',
   192→    kind: 'pending',
   193→  }
   194→}
   195→
   196→function statusLabelKey(status: string): string {
   197→  if (status === 'completed') return 'goal.status.completed'
   198→  if (status === 'on_hold') return 'goal.status.on_hold'
   199→  return 'common.active'
   200→}
   201→
   202→function statusToneClass(status: string): string {
   203→  if (status === 'completed') return 'border-income/30 text-income'
   204→  if (status === 'on_hold') return 'border-warning/30 text-warning'
   205→  return 'border-info/30 text-info'
   206→}

   290→    onSuccess: () => {
   291→      toast.success(t('credit.updated'))
   292→      qc.invalidateQueries({ queryKey: ['bank-credits'] })

   301→    mutationFn: (id: string) => api.del(`/api/bank-credits/${id}`),

   303→      toast.success(t('common.delete'))
   304→      qc.invalidateQueries({ queryKey: ['bank-credits'] })
   305→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   306→      qc.invalidateQueries({ queryKey: ['analytics'] })

   308→    },
   309→  })
   310→
   311→  const repaymentMutation = useMutation({
   312→    mutationFn: ({ id, body }: { id: string; body: Record<string, unknown> }) =>
   313→      api.post(`/api/bank-credits/${id}/repayments`, body),
   314→    onSuccess: () => {
   315→      toast.success(t('credit.repaymentAdded'))
   316→      qc.invalidateQueries({ queryKey: ['bank-credits'] })
   317→      qc.invalidateQueries({ queryKey: ['bank-credit', repayCredit?.id] })
   318→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   319→      qc.invalidateQueries({ queryKey: ['analytics'] })

   324→  // ---------------------------------------------------------------------------
   325→  // Handlers
   326→  // ---------------------------------------------------------------------------

   328→  const openCreate = () => {

   330→    setFormCreditName('')
   331→    setFormLender('')
   332→    setFormOriginalPrincipal('')
   333→    setFormDurationMonths('12')
   334→    setFormStartDate(todayLocal())
   335→    setFormMonthlyInstallment('')
   336→    setFormInterestRate('')

   338→    setFormPaymentDay('1')
   339→    setFormDescription('')
   340→    setFormStatus('active')
   341→    setCreditDialogOpen(true)
   342→  }
   343→
   344→  const openEdit = (c: BankCredit) => {
   345→    setEditing(c)
   346→    setFormCreditName(c.creditName)
   347→    setFormLender(c.lender)

   349→    setFormDurationMonths(String(c.durationMonths))

   355→    setFormDescription(c.description ?? '')

   358→  }

   363→      return

   365→    if (!formLender.trim()) {
   366→      toast.error(`${t('common.required')}: ${t('credit.lender')}`)
   367→      return
   368→    }

   370→    if (!Number.isFinite(principal) || principal <= 0) {
   371→      toast.error(`${t('common.required')}: ${t('credit.originalPrincipal')}`)
   372→      return
   373→    }
   374→    const duration = Number(formDurationMonths)

   377→      return
   378→    }
   379→    if (!formStartDate) {
   380→      toast.error(`${t('common.required')}: ${t('credit.startDate')}`)
   381→      return
   382→    }
   383→    const installment = Number(formMonthlyInstallment)
   384→    if (!Number.isFinite(installment) || installment <= 0) {
   385→      toast.error(`${t('common.required')}: ${t('credit.monthlyInstallment')}`)
   386→      return
   387→    }
   388→    const payDay = Number(formPaymentDay)
   389→    if (!Number.isFinite(payDay) || payDay < 1 || payDay > 31 || !Number.isInteger(payDay)) {
   390→      toast.error(`${t('common.required')}: ${t('credit.paymentDay')}`)
   391→      return
   392→    }
   393→
   394→    const body: Record<string, unknown> = {
   395→      creditName: formCreditName.trim(),
   396→      lender: formLender.trim(),
   397→      originalPrincipal: principal.toFixed(2),
   398→      durationMonths: duration,
   399→      startDate: formStartDate,

   401→      paymentDay: payDay,
   402→      status: formStatus,

  1108→function ScheduleGrid({
  1109→  query,
  1110→}: {
  1111→  query: ReturnType<typeof useQuery<BankCreditDetail>>
  1112→}) {
  1113→  const { t, date, money } = useI18n()
  1114→
  1115→  const installments = query.data?.installments ?? []
  1116→  const isLoading = query.isLoading
  1117→
  1118→  const paidLabel = useMemo(
  1119→    () => t('credit.paidInstallments').split(' ')[0] || 'Paid',
  1120→    [t],
  1121→  )
  1122→  const remainingLabel = t('credit.remainingInstallments')
  1123→  const overdueLabel = t('common.remaining')
  1124→
  1125→  // Group installments by year (ascending).
  1126→  const grouped = useMemo(() => {
  1127→    const map = new Map<number, CreditInstallment[]>()
  1128→    for (const inst of installments) {
  1129→      const d = new Date(inst.dueDate)
  1130→      const y = isNaN(d.getTime()) ? 0 : d.getFullYear()
  1131→      if (!map.has(y)) map.set(y, [])
  1132→      map.get(y)!.push(inst)
  1133→    }
  1134→    return Array.from(map.entries()).sort((a, b) => a[0] - b[0])
  1135→  }, [installments])
  1136→
  1137→  if (isLoading) {
  1138→    return (
  1139→      <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
  1140→        {Array.from({ length: 12 }).map((_, i) => (
  1141→          <Skeleton key={i} className="h-20 w-full" />
  1142→        ))}
  1143→      </div>
  1144→    )
  1145→  }
  1146→
  1147→  if (installments.length === 0) {
  1148→    return <EmptyState title={t('common.empty')} />
  1149→  }
  1150→
  1151→  return (
  1152→    <ScrollArea className="max-h-[55vh] pr-3">
  1153→      <div className="space-y-5 p-1">
  1154→        {/* Legend */}
  1155→        <div className="flex flex-wrap items-center gap-4 text-xs">
  1156→          <LegendDot color="rgba(34,197,94,0.55)" label={paidLabel} />
  1157→          <LegendDot color="rgba(239,68,68,0.55)" label={overdueLabel} />
  1158→          <LegendDot color="rgba(245,158,11,0.45)" label={remainingLabel} />
  1159→        </div>
  1160→
  1161→        {grouped.map(([year, items]) => (
  1162→          <div key={year} className="space-y-2">
  1163→            {grouped.length > 1 && (
  1164→              <div className="flex items-center gap-2 text-xs font-medium text-muted-foreground">
  1165→                <Flag className="h-3 w-3" />
  1166→                <span>
  1167→                  {t('common.year')} {year}
  1168→                </span>
  1169→                <div className="h-px flex-1 bg-border" />
  1170→              </div>
  1171→            )}
  1172→            <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
  1173→              {items.map((inst) => {
  1174→                const seg = installmentSegment(inst)
  1175→                const label =
  1176→                  inst.status === 'paid'
  1177→                    ? paidLabel
  1178→                    : seg.kind === 'overdue'
  1179→                      ? overdueLabel
  1180→                      : remainingLabel
  1181→                return (
