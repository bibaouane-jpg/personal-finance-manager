   144→   144→// Date helpers — produce YYYY-MM-DD strings the backend can parse via parseLocalDate.

   147→   147→function pad(n: number): string {
   148→   148→  return String(n).padStart(2, '0')
   149→   149→}
   150→   150→
   151→   151→function fmtDate(d: Date): string {
   152→   152→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
   153→   153→}
   154→   154→
   155→   155→function periodToRange(
   156→   156→  period: PeriodFilter,
   157→   157→  customFrom?: string,
   158→   158→  customTo?: string,
   159→   159→): { from?: string; to?: string } {
   160→   160→  const today = new Date()
   161→   161→  const y = today.getFullYear()
   162→   162→  const m = today.getMonth()
   163→   163→  const d = today.getDate()
   164→   164→  switch (period) {
   165→   165→    case 'today': {
   166→   166→      const t = fmtDate(today)
   167→   167→      return { from: t, to: t }
   168→   168→    }
   169→   169→    case 'this_week': {
   170→   170→      // Monday-based week (Sunday = end of week)
   171→   171→      const dow = today.getDay() // 0 = Sunday, 1 = Monday …
   172→   172→      const offset = dow === 0 ? -6 : 1 - dow
   173→   173→      const monday = new Date(y, m, d + offset)
   174→   174→      const sunday = new Date(y, m, d + offset + 6)
   175→   175→      return { from: fmtDate(monday), to: fmtDate(sunday) }
   176→   176→    }
   177→   177→    case 'this_month': {
   178→   178→      return { from: fmtDate(new Date(y, m, 1)), to: fmtDate(new Date(y, m + 1, 0)) }
   179→   179→    }
   180→   180→    case 'this_year': {
   181→   181→      return { from: `${y}-01-01`, to: `${y}-12-31` }
   182→   182→    }
   183→   183→    case 'all_time':
   184→   184→      return {}
   185→   185→    case 'custom_range':
   186→   186→      return {
   187→   187→        from: customFrom || undefined,
   188→   188→        to: customTo || undefined,
   189→   189→      }
   190→   190→  }
   191→   191→}
   192→   192→
   193→   193→function todayLocal(): string {
   194→   194→  return toLocalDateString(new Date())
   195→   195→}
   196→   196→
   197→   197→function clampPage(p: number, totalPages: number): number {
   198→   198→  if (p < 1) return 1
   199→   199→  if (p > totalPages) return totalPages
   200→   200→  return p
   201→   201→}
   202→   202→
   203→   203→function buildPageList(current: number, totalPages: number): (number | 'ellipsis')[] {
   204→   204→  if (totalPages <= 7) {
   205→   205→    return Array.from({ length: totalPages }, (_, i) => i + 1)
   206→   206→  }
   207→   207→  const pages: (number | 'ellipsis')[] = [1]
   208→   208→  const left = Math.max(2, current - 1)
   209→   209→  const right = Math.min(totalPages - 1, current + 1)
   210→   210→  if (left > 2) pages.push('ellipsis')
   211→   211→  for (let i = left; i <= right; i++) pages.push(i)
   212→   212→  if (right < totalPages - 1) pages.push('ellipsis')
   213→   213→  pages.push(totalPages)
   214→   214→  return pages
   215→   215→}
