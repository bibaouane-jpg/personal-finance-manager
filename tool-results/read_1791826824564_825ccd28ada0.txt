   304→function safeColor(c: string | null | undefined, fallback = MUTED_COLOR): string {
   305→  if (!c) return fallback
   306→  return c.startsWith('#') || c.startsWith('var(') || c.startsWith('rgb') ? c : fallback
   307→}

   309→function formatMonthLabel(month: string, locale: 'en' | 'fr'): string {
   310→  // month comes as "YYYY-MM"
   311→  const d = new Date(`${month}-01T00:00:00Z`)
   312→  if (isNaN(d.getTime())) return month
   313→  return new Intl.DateTimeFormat(locale === 'fr' ? 'fr-FR' : 'en-US', {
   314→    month: 'short',
   315→    year: '2-digit',
   316→    timeZone: 'UTC',
   317→  }).format(d)
   318→}
   319→
   320→function formatTrendDate(iso: string, locale: 'en' | 'fr'): string {
   321→  const d = new Date(iso)
   322→  if (isNaN(d.getTime())) return iso
   323→  return new Intl.DateTimeFormat(locale === 'fr' ? 'fr-FR' : 'en-US', {
   324→    month: 'short',
   325→    day: 'numeric',
   326→  }).format(d)
   327→}
   328→
   329→function formatYearLabel(year: number, locale: 'en' | 'fr'): string {
   330→  return new Intl.DateTimeFormat(locale === 'fr' ? 'fr-FR' : 'en-US', {
   331→    year: 'numeric',
   332→  }).format(new Date(Date.UTC(year, 0, 1)))
   333→}

   335→function asNumber(v: string | number | null | undefined): number {
   336→  if (v === null || v === undefined) return 0
   337→  const n = typeof v === 'number' ? v : Number(v)
   338→  return Number.isFinite(n) ? n : 0
   339→}

   352→function MoneyTooltip({
   353→  active,
   354→  payload,
   355→  label,
   356→  money,
   357→  labelFormatter,
   358→}: {
   359→  active?: boolean
   360→  payload?: TooltipEntry[]
   361→  label?: string | number
   362→  money: (v: string | number | null | undefined) => string
   363→  labelFormatter?: (label: string | number | undefined) => string
   364→}) {
   365→  if (!active || !payload || payload.length === 0) return null
   366→  return (
   367→    <div className="rounded-md border bg-popover p-3 text-xs shadow-md">
   368→      {label !== undefined && label !== '' && (
   369→        <p className="mb-1 font-medium text-foreground">
   370→          {labelFormatter ? labelFormatter(label) : label}
   371→        </p>
   372→      )}
   373→      <div className="space-y-1">
   374→        {payload.map((entry, i) => (
   375→          <div key={i} className="flex items-center gap-2">
   376→            <span
   377→              className="inline-block h-2.5 w-2.5 rounded-full"
   378→              style={{ backgroundColor: entry.color || MUTED_COLOR }}
   379→            />
   380→            <span className="text-muted-foreground">{entry.name}:</span>
   381→            <span className="font-medium tabular-nums text-foreground">
   382→              {money(entry.value)}
   383→            </span>
   384→          </div>
   385→        ))}
   386→      </div>
   387→    </div>
   388→  )
   389→}

   550→function IncomeSection({ data, loading }: { data?: IncomeData; loading: boolean }) {
   551→  const { t, money, locale } = useI18n()
   552→  const noData = t('dash.noData')
   553→
   554→  const monthly = useMemo(
   555→    () =>
   556→      (data?.monthly ?? []).map((m) => ({
   557→        month: m.month,
   558→        amount: asNumber(m.amount),
   559→      })),
   560→    [data?.monthly]
   561→  )
   562→  const byCategory = useMemo(
   563→    () =>
   564→      (data?.byCategory ?? []).map((c) => ({
   565→        name: c.name,
   566→        value: asNumber(c.amount),
   567→        color: safeColor(c.color),
   568→      })),
   569→    [data?.byCategory]
   570→  )
   571→  const byArticle = useMemo(
   572→    () =>
   573→      (data?.byArticle ?? []).map((a) => ({
   574→        name: a.name,
   575→        amount: asNumber(a.amount),
   576→      })),
   577→    [data?.byArticle]
   578→  )
   579→  const byWallet = useMemo(
   580→    () =>
   581→      (data?.byWallet ?? []).map((w) => ({
   582→        name: w.name,
   583→        amount: asNumber(w.amount),
   584→        color: safeColor(w.color),
   585→      })),
   586→    [data?.byWallet]
   587→  )
   588→
   589→  const concentration = data?.sourceConcentration ?? 0
   590→
   591→  if (loading) return <AnalyticsSkeleton />
   592→
   593→  return (
   594→    <div className="space-y-4">
   595→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   596→        <StatCard
   597→          label={t('common.total')}
   598→          value={money(data?.total)}
   599→          icon={TrendingUp}
   600→          tone="income"
   601→        />
   602→        <StatCard
   603→          label={t('analytics.average')}
   604→          value={money(data?.average)}
   605→          icon={CalendarClock}
   606→          tone="default"
   607→        />
   608→        <StatCard
   609→          label={t('analytics.largest')}
   610→          value={data?.largest ? money(data.largest.amount) : money('0')}
   611→          hint={data?.largest?.name ?? data?.largest?.categoryName ?? undefined}
   612→          icon={Trophy}
   613→          tone="info"
   614→        />
   615→        <StatCard
   616→          label={t('analytics.savingsRate')}
   617→          value={`${concentration.toFixed(1)}%`}
   618→          hint={data?.byCategory?.[0]?.name ?? undefined}
   619→          icon={Percent}
   620→          tone="warning"
   621→        />
   622→      </div>
   623→
   624→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   625→        <ChartCard title={t('analytics.monthlyTrend')}>
   626→          {monthly.length === 0 ? (
   627→            <ChartEmpty noData={noData} />
   628→          ) : (
   629→            <ResponsiveContainer width="100%" height={260}>
   630→              <AreaChart data={monthly} margin={{ top: 8, right: 8, left: 8, bottom: 0 }}>
   631→                <defs>
   632→                  <linearGradient id="incomeGrad" x1="0" y1="0" x2="0" y2="1">
   633→                    <stop offset="0%" stopColor={INCOME_COLOR} stopOpacity={0.35} />
   634→                    <stop offset="100%" stopColor={INCOME_COLOR} stopOpacity={0} />
   635→                  </linearGradient>
   636→                </defs>
   637→                <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   638→                <XAxis
   639→                  dataKey="month"
   640→                  tickFormatter={(m) => formatMonthLabel(String(m), locale)}
   641→                  tick={{ fontSize: 11 }}
   642→                  stroke="var(--muted-foreground)"
   643→                />
   644→                <YAxis
   645→                  tickFormatter={(v) => money(v as number)}
   646→                  tick={{ fontSize: 11 }}
   647→                  stroke="var(--muted-foreground)"
   648→                  width={90}
   649→                />
   650→                <RechartsTooltip
   651→                  content={(props) => (
   652→                    <MoneyTooltip
   653→                      {...props}
   654→                      money={money}
   655→                      labelFormatter={(l) => formatMonthLabel(String(l), locale)}
   656→                    />
   657→                  )}
   658→                />
   659→                <Area
   660→                  type="monotone"

  1920→function SectionLoader({ section, filter, from, to }: SectionProps) {
  1921→  const api = useApi()
  1922→  const { t } = useI18n()
  1923→  const noData = t('dash.noData')
  1924→
  1925→  const enabled = filter !== 'custom_range' || (!!from && !!to)
  1926→
  1927→  const queryKey = useMemo(
  1928→    () => ['analytics', filter, from, to, section] as const,
  1929→    [filter, from, to, section]
  1930→  )
  1931→
  1932→  const { data, isLoading } = useQuery<Record<string, unknown>>({
  1933→    queryKey,
  1934→    queryFn: () =>
  1935→      api.get<Record<string, unknown>>('/api/analytics', {
  1936→        query: {
  1937→          filter,
  1938→          ...(filter === 'custom_range' && from ? { from } : {}),
  1939→          ...(filter === 'custom_range' && to ? { to } : {}),
  1940→          section,
  1941→        },
  1942→      }),
  1943→    enabled,
  1944→  })
  1945→
  1946→  const sectionData = data?.[section]
  1947→
  1948→  if (!enabled) {
  1949→    return (
  1950→      <div className="flex h-[200px] items-center justify-center">
  1951→        <EmptyState title={t('common.apply')} description={`${t('common.from')} / ${t('common.to')}`} />
  1952→      </div>
  1953→    )
  1954→  }
  1955→
  1956→  if (isLoading) return <AnalyticsSkeleton />
  1957→  if (!sectionData) {
  1958→    return (
  1959→      <div className="flex h-[200px] items-center justify-center">
  1960→        <EmptyState title={noData} />
  1961→      </div>
  1962→    )
  1963→  }
  1964→
  1965→  switch (section) {
  1966→    case 'income':
  1967→      return <IncomeSection data={sectionData as IncomeData} loading={false} />
  1968→    case 'expenses':
  1969→      return <ExpensesSection data={sectionData as ExpenseData} loading={false} />
  1970→    case 'wallets':
  1971→      return <WalletsSection data={sectionData as WalletsData} loading={false} />
  1972→    case 'cash_withdrawals':
  1973→      return <CashWithdrawalsSection data={sectionData as CashWithdrawalsData} loading={false} />
  1974→    case 'transfers':
  1975→      return <TransfersSection data={sectionData as TransfersData} loading={false} />
  1976→    case 'net_cash_flow':
  1977→      return <NetCashFlowSection data={sectionData as NetCashFlowData} loading={false} />
  1978→    case 'allocations':
  1979→      return <AllocationsSection data={sectionData as AllocationsData} loading={false} />
  1980→    case 'bank_credits':
  1981→      return <BankCreditsSection data={sectionData as BankCreditsData} loading={false} />
  1982→    case 'personal_loans':
  1983→      return <PersonalLoansSection data={sectionData as PersonalLoansData} loading={false} />
  1984→    case 'goals':
  1985→      return <GoalsSection data={sectionData as GoalsData} loading={false} />
  1986→    default:
  1987→      return (
  1988→        <div className="flex h-[200px] items-center justify-center">
  1989→          <EmptyState title={noData} />
