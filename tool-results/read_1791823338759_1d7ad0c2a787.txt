   135→   135→function todayLocal(): string {

   139→   139→function startOfToday(): Date {

   145→   145→/** Map a repayment progress percent to a credit-card color palette.
   146→   146→ *  <50% red, 50-<70% orange, 70-<90% dark green, 90-100% green. */
   147→   147→function creditCardColor(progressPct: number): {
   148→   148→  stripe: string
   149→   149→  text: string
   150→   150→  tint: string
   151→   151→} {
   152→   152→  if (progressPct < 50) {
   153→   153→    return { stripe: '#ef4444', text: '#b91c1c', tint: 'rgba(239,68,68,0.08)' }
   154→   154→  }
   155→   155→  if (progressPct < 70) {
   156→   156→    return { stripe: '#f97316', text: '#c2410c', tint: 'rgba(249,115,22,0.08)' }
   157→   157→  }
   158→   158→  if (progressPct < 90) {
   159→   159→    return { stripe: '#15803d', text: '#15803d', tint: 'rgba(21,128,61,0.10)' }
   160→   160→  }
   161→   161→  return { stripe: '#22c55e', text: '#15803d', tint: 'rgba(34,197,94,0.10)' }
   162→   162→}
   163→   163→
   164→   164→/** Classify an installment segment color for the schedule grid. */
   165→   165→function installmentSegment(inst: CreditInstallment): {
   166→   166→  bg: string
   167→   167→  border: string
   168→   168→  text: string
   169→   169→  kind: 'paid' | 'overdue' | 'pending'
   170→   170→} {
   171→   171→  if (inst.status === 'paid') {
   172→   172→    return {
   173→   173→      bg: 'rgba(34,197,94,0.15)',
   174→   174→      border: 'rgba(34,197,94,0.55)',
   175→   175→      text: '#15803d',
   176→   176→      kind: 'paid',
   177→   177→    }
   178→   178→  }
   179→   179→  const due = new Date(inst.dueDate)
   180→   180→  if (!isNaN(due.getTime()) && due < startOfToday()) {
   181→   181→    return {
   182→   182→      bg: 'rgba(239,68,68,0.15)',
   183→   183→      border: 'rgba(239,68,68,0.55)',
   184→   184→      text: '#b91c1c',
   185→   185→      kind: 'overdue',
   186→   186→    }
   187→   187→  }
   188→   188→  return {
   189→   189→    bg: 'rgba(245,158,11,0.15)',
   190→   190→    border: 'rgba(245,158,11,0.45)',
   191→   191→    text: '#a16207',
   192→   192→    kind: 'pending',
   193→   193→  }
   194→   194→}
   195→   195→
   196→   196→function statusLabelKey(status: string): string {
   197→   197→  if (status === 'completed') return 'goal.status.completed'
   198→   198→  if (status === 'on_hold') return 'goal.status.on_hold'
   199→   199→  return 'common.active'
   200→   200→}
   201→   201→
   202→   202→function statusToneClass(status: string): string {
   203→   203→  if (status === 'completed') return 'border-income/30 text-income'
   204→   204→  if (status === 'on_hold') return 'border-warning/30 text-warning'
   205→   205→  return 'border-info/30 text-info'
   206→   206→}

  1118→  1118→  const paidLabel = useMemo(
